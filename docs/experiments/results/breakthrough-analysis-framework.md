                'learning_efficiency': self.calculate_learning_efficiency(domain_data)
            }

        return learning_curves

    def identify_breakthrough_moments(self, domain_data):
        """Identify moments of significant breakthrough in domain development"""
        breakthrough_moments = []

        for i in range(1, len(domain_data)):
            current_performance = domain_data[i]['performance_metric']
            previous_performance = domain_data[i-1]['performance_metric']

            # Detect significant performance jumps
            improvement_ratio = current_performance / previous_performance if previous_performance > 0 else float('inf')

            if improvement_ratio > 1.5:  # 50% improvement threshold
                breakthrough_moments.append({
                    'timestamp': domain_data[i]['timestamp'],
                    'improvement_ratio': improvement_ratio,
                    'performance_before': previous_performance,
                    'performance_after': current_performance,
                    'breakthrough_magnitude': improvement_ratio - 1.0
                })

        return breakthrough_moments
